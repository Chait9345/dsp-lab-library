%---------------------------------------------------------------
% EXAMPLE Ex-1: Unit step, delayed step (delay 5), step with amplitude 5,
% rectangular window of length 5, ramp. (n = 0:15)
%---------------------------------------------------------------
clc; clear; close all;
n = 0:15;
u = (n>=0); % unit step
u5 = (n-5)>=0; % delayed step

subplot(3,2,2); stem(n,u5); title('Delayed step');

subplot(3,2,5); stem(n,xr); title('Ramp');

% Diff. equation -> b, a -> response (the core of this lab)

% Impulse response (two ways)

[h,n] = impz(b,a,N); % or direct

% With initial conditions
ic = filtic(b,a,[y(-1) y(-2)],[x(-1) x(-2)]);
y = filter(b,a,x,ic);

% Analysis toolkit
[r,p,k] = residuez(b,a); % partial fractions
[z,p,k] = tf2zp(b,a); % zeros, poles, gain
[b,a] = zp2tf(z,p,k); % back to b, a
zplane(b,a); % pole-zero plot
[h,w] = freqz(b,a); % frequency response

%---------------------------------------------------------------
% EXERCISE Q2: Anti-causal system H(z) = (z^2-1)/(z-3)^2. Find
% (a) impulse response (b) difference equation (c) pole-zero plot
% (d) output for x(n) = 3sin(pi*n/3)u(n).
%---------------------------------------------------------------
clc; clear; close all;
% divide num & den by z^2: H = (1 - z^-2)/(1 - 6z^-1 + 9z^-2)
b = [1 0 -1]; a = [1 -6 9];

[g,m] = impz(fliplr(b),fliplr(a),20);
h = flipud(g)'; nh = -19:0; % h[n] for n<=0

% (d)
n = 0:50; x = 3*sin(n*pi/3);
y = conv(x,h); ny = (n(1)+nh(1)):(n(end)+nh(end));
figure; stem(ny,y); title('Output');

# EXPERIMENT 4: IIR FILTER DESIGN (MATLAB)
#####################################################################

%---------------------------------------------------------------
% Patterns
%---------------------------------------------------------------

% THE IIR template (swap the 2 function names)
[N,wc] = buttord(wp,ws,rp,rs,'s'); % order
[b,a] = butter(N,wc,'s'); % coefficients
[h,w] = freqs(b,a); % analog response
semilogx(w/(2*pi),20*log10(abs(h)));

% Family table
% buttord  -> butter(N,wc,'s')
% cheb1ord -> cheby1(N,rp,wc,'s')
% cheb2ord -> cheby2(N,rs,wc,'s')
% ellipord -> ellip(N,rp,rs,wc,'s')
% High-pass: add 'high' -> butter(N,wc,'high','s')

% Analog -> digital (impulse invariance)
[bz,az] = impinvar(bs,as); % T = 1
[h,w] = freqz(bz,az);
plot(w/pi,20*log10(abs(h)));

% Direct digital filter (Fs given)
[b,a] = butter(3, 100/(Fs/2)); % fc normalized
y = filter(b,a,x);


%---------------------------------------------------------------
% EXAMPLE Ex-1 (Analog): 5th-order Butterworth prototype, then scale to
% cutoff 0.5*pi rad/s.
%---------------------------------------------------------------
clc; clear; close all;
N = 5;
[z,p,k] = buttap(N); % prototype (wc = 1)
[num,den] = zp2tf(z,p,k);
[h,w] = freqs(num,den);
semilogx(w,20*log10(abs(h))); title('Prototype');
wc = 0.5*pi;
p1 = p*wc; k1 = k*wc^N; % scale poles and gain
[num1,den1] = zp2tf(z,p1,k1)
[h1,w1] = freqs(num1,den1);
figure; semilogx(w1,20*log10(abs(h1))); title('Scaled filter');


%---------------------------------------------------------------
% EXAMPLE Ex-2 (Analog): Butterworth LPF, fp = 100 Hz, fs = 500 Hz,
% Rp = 1 dB, Rs = 40 dB.
%---------------------------------------------------------------
clc; clear; close all;
wp = 2*pi*100; ws = 2*pi*500; rp = 1; rs = 40;
[N,wc] = buttord(wp,ws,rp,rs,'s') % N = 4
[b,a] = butter(N,wc,'s');
[h,w] = freqs(b,a);
semilogx(w/(2*pi),20*log10(abs(h))); grid on;
xlabel('Hz'); ylabel('dB');


%---------------------------------------------------------------
% EXERCISE Q1 (Analog): fp = 200 Hz, Rp = 1 dB, fs = 700 Hz, Rs = 50 dB.
% Design LPF: Butterworth, Cheby-I, Cheby-II, Elliptic. Compare.
%---------------------------------------------------------------
clc; clear; close all;
wp = 2*pi*200; ws = 2*pi*700; rp = 1; rs = 50;
[N1,w1] = buttord(wp,ws,rp,rs,'s'); [b1,a1] = butter(N1,w1,'s');
[N2,w2] = cheb1ord(wp,ws,rp,rs,'s'); [b2,a2] = cheby1(N2,rp,w2,'s');
[N3,w3] = cheb2ord(wp,ws,rp,rs,'s'); [b3,a3] = cheby2(N3,rs,w3,'s');
[N4,w4] = ellipord(wp,ws,rp,rs,'s'); [b4,a4] = ellip(N4,rp,rs,w4,'s');

/*---------------------------------------------------------------
 EXAMPLE Exp 7 Example: N-point DFT of x = (1 1 2 -2 3 0 5 1).
 Expected: X[0] = 11, X[1] = 0.828 + j4.414, X[2] = -3 - j2, X[4] = 11
 (check with MATLAB: fft([1 1 2 -2 3 0 5 1])).
 ---------------------------------------------------------------*/
#include <stdio.h>
#include <math.h>
#define PI 3.14159265
#define N 8
float x[N] = {1, 1, 2, -2, 3, 0, 5, 1};
float Xr[N], Xi[N];

void main()
{
 int k, n;
 for (k = 0; k < N; k++) {
 Xr[k] = 0; Xi[k] = 0;
 for (n = 0; n < N; n++) {
 Xr[k] += x[n]*cos(2*PI*k*n/N);
 Xi[k] -= x[n]*sin(2*PI*k*n/N);
 }
 printf("X[%d] = %7.3f + j(%7.3f)\n", k, Xr[k], Xi[k]);
 }
}


/*---------------------------------------------------------------
 EXERCISE Exp 7 Exercise: N-point IDFT of (1 1 2 -2 3 0 5 1).
 Only 2 changes from the DFT: +sin and /N.
 Check: ifft([1 1 2 -2 3 0 5 1]). x[0] = 1.375.
 ---------------------------------------------------------------*/
#include <stdio.h>
#include <math.h>
#define PI 3.14159265
#define N 8
float X[N] = {1, 1, 2, -2, 3, 0, 5, 1};
float xr[N], xi[N];

void main()
{
 int k, n;
 for (n = 0; n < N; n++) {
 xr[n] = 0; xi[n] = 0;
 for (k = 0; k < N; k++) {
 xr[n] += X[k]*cos(2*PI*k*n/N);
 xi[n] += X[k]*sin(2*PI*k*n/N); /* + sign for IDFT */
 }
 xr[n] /= N; xi[n] /= N; /* divide by N */
 printf("x[%d] = %7.3f + j(%7.3f)\n", n, xr[n], xi[n]);
 }
}

/*---------------------------------------------------------------
 EXERCISE Q1: Echo effect.
 ---------------------------------------------------------------*/
#include "L138_aic3106_init.h"
#define BUF_SIZE 16000 /* 0.33 s at 48 kHz */
#define GAIN 0.6 /* decay of each echo */
int16_t buffer[BUF_SIZE];
int i = 0;

interrupt void interrupt4(void)
{
 int16_t input, delayed, output;
 input = input_left_sample();
 delayed = buffer[i];
 output = input + delayed;
 buffer[i] = input + delayed*GAIN; /* FEEDBACK -> repeating echoes */
 i = (i + 1) % BUF_SIZE;
 output_left_sample(output);
 return;
}

int main(void)
{
 L138_initialise_intr(FS_48000_HZ, ADC_GAIN_0DB, DAC_ATTEN_0DB);
 while (1);
}


/*---------------------------------------------------------------
